Thuật toán Minimax + Alpha-Beta + Heuristic 5×5 
// Hàm Minimax kết hợp Cắt tỉa Alpha-Beta cho bàn cờ 5x5
function minimaxAlphaBeta(board, depth, alpha, beta, isMaximizing) {
    nodesEvaluated++;
    const terminalScore = checkTerminalOrHeuristic(board, depth);
    if (terminalScore !== null) return terminalScore;

    // Sắp xếp nước đi (Move Ordering) ưu tiên ô trung tâm & kề cận
    const moves = getOrderedMoves(board);

    if (isMaximizing) { // Lượt của AI (O) - Tối đa hóa điểm
        let maxEval = -Infinity;
        for (const move of moves) {
            board[move] = AI_PLAYER;
            const evaluation = minimaxAlphaBeta(board, depth - 1, alpha, beta, false);
            board[move] = EMPTY;

            if (evaluation > maxEval) maxEval = evaluation;
            if (evaluation > alpha) alpha = evaluation;

            // Cắt tỉa nhánh Beta (Beta Cutoff)
            if (beta <= alpha) {
                prunedBranches++;
                break;
            }
        }
        return maxEval;
    } else { // Lượt của Người chơi (X) - Tối thiểu hóa điểm của AI
        let minEval = Infinity;
        for (const move of moves) {
            board[move] = HUMAN_PLAYER;
            const evaluation = minimaxAlphaBeta(board, depth - 1, alpha, beta, true);
            board[move] = EMPTY;

            if (evaluation < minEval) minEval = evaluation;
            if (evaluation < beta) beta = evaluation;

            // Cắt tỉa nhánh Alpha (Alpha Cutoff)
            if (beta <= alpha) {
                prunedBranches++;
                break;
            }
        }
        return minEval;
    }
}
Tại sao bàn cờ 5×5 cần Cắt tỉa Alpha-Beta & Move Ordering?

Bàn cờ 5×5 có 25 ô ($25! \approx 1.55 \times 10^{25}$ trạng thái). Nếu dùng Minimax thuần sẽ bị bùng nổ tổ hợp. Nhờ Cắt tỉa Alpha-Beta kết hợp Sắp xếp thứ tự ưu tiên nước đi (Move Ordering) và Hàm lượng giá chuỗi mở/nửa mở, AI có thể loại bỏ tới 85% – 96% số nhánh thừa mà vẫn tìm ra nước đi tối ưu trong vài mili-giây!
